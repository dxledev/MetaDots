cmake_minimum_required(VERSION 3.22)
project(MetaDots VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(METADOTS_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk4>=4.18)
pkg_check_modules(TAGLIB REQUIRED IMPORTED_TARGET taglib>=2.0)
find_package(nlohmann_json 3.10 REQUIRED)
add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(METADOTS_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(metadots_core src/core/theme.cpp src/core/metadata.cpp src/core/gtk_config.cpp src/core/settings.cpp)
target_include_directories(metadots_core PUBLIC src)
target_link_libraries(metadots_core PUBLIC nlohmann_json::nlohmann_json PkgConfig::TAGLIB)
add_executable(metadots src/main.cpp src/cli.cpp src/ui/window.cpp src/ui/blur_bin.cpp)
target_link_libraries(metadots PRIVATE metadots_core PkgConfig::GTK)
include(CTest)
if(BUILD_TESTING)
  add_executable(theme_tests tests/theme_tests.cpp)
  target_link_libraries(theme_tests PRIVATE metadots_core)
  target_compile_options(theme_tests PRIVATE -UNDEBUG)
  add_test(NAME theme COMMAND theme_tests)
  add_executable(gtk_config_tests tests/gtk_config_tests.cpp)
  target_link_libraries(gtk_config_tests PRIVATE metadots_core)
  target_compile_options(gtk_config_tests PRIVATE -UNDEBUG)
  add_test(NAME gtk_config COMMAND gtk_config_tests)
  add_executable(settings_tests tests/settings_tests.cpp)
  target_link_libraries(settings_tests PRIVATE metadots_core)
  target_compile_options(settings_tests PRIVATE -UNDEBUG)
  add_test(NAME settings COMMAND settings_tests)
  add_executable(metadata_tests tests/metadata_tests.cpp)
  target_link_libraries(metadata_tests PRIVATE metadots_core)
  target_compile_options(metadata_tests PRIVATE -UNDEBUG)
  add_test(NAME metadata COMMAND metadata_tests)
  add_test(NAME cli COMMAND bash ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_test.sh $<TARGET_FILE:metadots>)
  add_test(NAME install COMMAND bash ${CMAKE_CURRENT_SOURCE_DIR}/tests/install_test.sh $<TARGET_FILE:metadots>)
endif()
install(TARGETS metadots RUNTIME DESTINATION bin)
install(FILES assets/io.github.metadots.desktop DESTINATION share/applications)
install(FILES assets/io.github.metadots.svg DESTINATION share/icons/hicolor/scalable/apps)
install(FILES assets/io.github.metadots-symbolic.svg DESTINATION share/icons/hicolor/symbolic/apps)
